"use client";

/* eslint-disable react-hooks/exhaustive-deps */
import React, { useState, useEffect, useCallback } from "react";
import { jwtDecode } from "jwt-decode";
import { AuthContext } from "./auth-context";
import { UserData, AuthContextType } from "./auth-types";
import { useRouter } from "next/navigation";

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [isAuthenticated, setIsAuthenticated] = useState(false);
  const [user, setUser] = useState<UserData | null>(null);
  const [companyLogo, setCompanyLogo] = useState("");
  const router = useRouter();

  const logout = useCallback(() => {
    localStorage.removeItem("beamlmsUserToken");
    setUser(null);
    setIsAuthenticated(false);
    router.push("/login");
  }, [router]);

  const validateToken = useCallback(() => {
    const token = localStorage.getItem("beamlmsUserToken");
    if (!token) {
      logout();
      return;
    }

    try {
      const decoded: any = jwtDecode(token);

      // 🔴 Token expired
      if (decoded.exp && decoded.exp * 1000 < Date.now()) {
        logout();
        return;
      }

      setUser({
        id:
          decoded["http://schemas.xmlsoap.org/ws/2005/05/identity/claims/nameidentifier"] ||
          "",
        firstName:
          decoded["http://schemas.xmlsoap.org/ws/2005/05/identity/claims/givenname"],
        lastName:
          decoded["http://schemas.xmlsoap.org/ws/2005/05/identity/claims/surname"],
        email:
          decoded["http://schemas.xmlsoap.org/ws/2005/05/identity/claims/emailaddress"] ||
          "",
        role:
          decoded["http://schemas.microsoft.com/ws/2008/06/identity/claims/role"] ||
          "",
        accountType: decoded.accountType,
        userName:
          decoded["http://schemas.xmlsoap.org/ws/2005/05/identity/claims/name"],
      });

      setCompanyLogo(localStorage.getItem("companyLogo") || "");
      setIsAuthenticated(true);
    } catch (err) {
      console.error("Invalid token:", err);
      logout();
    }
  }, [logout]);

  // 🔹 Run once on app load
  useEffect(() => {
    validateToken();
  }, []);

  // 🔹 Optional: auto logout when token expires
  useEffect(() => {
    const interval = setInterval(validateToken, 60 * 1000); // every 1 min
    return () => clearInterval(interval);
  }, [validateToken]);

  const login = (token: string) => {
    localStorage.setItem("beamlmsUserToken", token);
    validateToken();
  };

  const contextValue: AuthContextType = {
    isAuthenticated,
    companyLogo,
    user,
    login,
    logout,
  };

  return (
    <AuthContext.Provider value={contextValue}>
      {children}
    </AuthContext.Provider>
  );
}
